# Default group. The major AI assistant and search crawlers are listed
# explicitly alongside `*` so our policy for them is unambiguous; they inherit
# the exact same Allow/Disallow rules below (no special access).
User-agent: *
User-agent: GPTBot
User-agent: OAI-SearchBot
User-agent: ChatGPT-User
User-agent: ClaudeBot
User-agent: Claude-Web
User-agent: anthropic-ai
User-agent: PerplexityBot
User-agent: meta-externalagent
User-agent: Amazonbot
User-agent: cohere-ai
User-agent: YouBot

# Content usage preferences (contentsignals.org): allow search indexing and AI
# answering with attribution, but signal no training on writers' content.
Content-Signal: search=yes, ai-input=yes, ai-train=no

Sitemap: https://paragraph.com/sitemap.xml

# All posts with unconfigured (default) accounts
# eg paragraph.com/@colin@paragraph.com
# or a wallet address
Disallow: /@*@*.*$

# All welcome posts
Disallow: /*welcome-to-paragraph*
Disallow: /*welcome-to-papyrus*

# The iFrame embeds.
Disallow: /*embed$
Disallow: /*memberships$

# Allow social media crawlers to access OG image generator
# This is critical for Twitter/Facebook/LinkedIn previews
Allow: /api/og
Allow: /api/og/*
# Allow public feeds and metadata endpoints that external bots rely on
Allow: /api/blogs/rss/
Allow: /api/published_notes/
Allow: /api/published_notes/*/html
Allow: /api/coins/metadata/
Allow: /api/coins/metadata/writer/
Allow: /api/share_img
# Disallow all other API calls
# eg /api/amp, /api/event
Disallow: /api*$

# Disallow admin and internal pages
Disallow: /admin*
Disallow: /internal*
Disallow: /dashboard*
Disallow: /editor*
Disallow: /settings*
Disallow: /setup*

# Disallow temporary and utility pages. Draft lists live under /dashboard
# (disallowed above) and post previews under /<blog>/preview/<id>, so there is
# deliberately no /*draft* rule — it would block /features/draft and any
# published post whose slug contains "draft".
Disallow: /refresh*
Disallow: /*preview*

# Disallow collector and NFT pages
Disallow: */collectors/*
Disallow: */nft/*
